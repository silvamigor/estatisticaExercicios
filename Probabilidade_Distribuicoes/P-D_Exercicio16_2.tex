\definevalorquestao

\begin{exercicioBanco}[\valorquestao]
As notas de um exame aplicado a um grande número de candidatos seguem aproximadamente uma distribuição normal com média de 70 pontos e desvio padrão de 10 pontos.
%
\begin{enumerate}[(a)]
    \item Qual a probabilidade de um candidato obter nota entre 60 e 75 pontos?
    %
    \item A partir de que nota um candidato está entre os 5\% melhores?
\end{enumerate}
%
\end{exercicioBanco}
