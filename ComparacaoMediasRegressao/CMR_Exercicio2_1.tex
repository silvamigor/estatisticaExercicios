\begin{exercicioBanco}[\valorquestao]

    \begin{center}
    \begin{tabular}{|c|cccccccc|}
        \hline
        \textbf{População} & \multicolumn{8}{c|}{\textbf{Dados}} \\
        \hline
        I  & 12 & 14 & 15 & 14 & 13 & 17 & 14 & 13 \\
        II & 13 & 17 & 14 & 13 & 16 & 17 & 18 & 16 \\
        \hline
    \end{tabular}
    \end{center}
\end{exercicioBanco}
