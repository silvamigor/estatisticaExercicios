\definevalorquestao

\begin{exercicioBanco}[\valorquestao]
Sejam \(A\) e \(B\) eventos de um espaço amostral. Responda cada uma das perguntas a seguir.
%
\begin{enumerate}[(a)]
    \item Se \(P(A) = \dfrac{1}{3}\) e \(P(B \mid A) = \dfrac{3}{5}\), então \(A\) e \(B\) são disjuntos?
    %
    \item Se \(P(A) = 0,1\), \(P(B) = 0,2\) e \(P(A\cap B) = 0,02\), então \(A\) e \(B\) são independentes?
\end{enumerate}
%
\end{exercicioBanco}
