\definevalorquestao

\begin{exercicioBanco}[\valorquestao]
Considere dois eventos disjuntos \(A\) e \(B\), com \(P(A) = 0,3\) e \(P(B) = 0,5\). Calcule:
%
\begin{multicols}{3}
    \begin{enumerate}[(a)]
        \item \(P(A \cap B)\);
        %
        \item \(P(A \cup B)\);
        %
        \item \(P(A \mid B)\);
        %
        \item \(P(A^{\complement})\);
        %
        \item \(P\big((A \cup B)^{\complement}\big)\).
    \end{enumerate}
\end{multicols}
%
\end{exercicioBanco}
